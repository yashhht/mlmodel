import React,{useEffect,useMemo,useRef,useState} from "react";
import {createRoot} from "react-dom/client";
import {Activity,AlertTriangle,ArrowLeft,ArrowRight,BrainCircuit,CheckCircle2,ChevronRight,Cloud,CloudRain,Database,Gauge,Leaf,MapPin,Mic,Network,Radio,Search,ShieldAlert,ShieldCheck,Thermometer,Users,Volume2,Wind,X,Plane} from "lucide-react";
import {ResponsiveContainer,LineChart,Line,XAxis,YAxis,CartesianGrid,Tooltip} from "recharts";
import "./styles.css";

const WS="ws://localhost:8000/ws";
const MODES=[["normal","Normal"],["spike","Sensor Spike"],["drift","Sensor Drift"],["stuck","Stuck Sensor"],["dropout","Communication Gap"],["extreme","Weather Event"]];

function App(){
 const [page,setPage]=useState("home");
 const [city,setCity]=useState("Pune");
 const [station,setStation]=useState("AWS_001");
 const [connected,setConnected]=useState(false);
 const [mode,setMode]=useState("normal");
 const [tab,setTab]=useState("overview");
 const [search,setSearch]=useState("");
 const [cities,setCities]=useState([]);
 const [stations,setStations]=useState([]);
 const [latest,setLatest]=useState({});
 const [history,setHistory]=useState({});
 const [events,setEvents]=useState({});
 const [nearby,setNearby]=useState([]);
 const [departments,setDepartments]=useState(null);
 const [voice,setVoice]=useState(false);
 const ws=useRef(null);
 const cityStrip=useRef(null);
 const recognition=useRef(null);

 useEffect(()=>{
   fetch("http://localhost:8000/api/cities")
     .then(r=>r.json())
     .then(x=>setCities(x.cities||[]))
     .catch(()=>{});
   fetch("http://localhost:8000/api/stations")
     .then(r=>r.json())
     .then(x=>setStations(x.stations||[]))
     .catch(()=>{});
 },[]);

 useEffect(()=>{
   if(page!=="dashboard") return;
   connect();
   return()=>ws.current?.close();
 },[page]);

 function connect(){
   const s=new WebSocket(WS);
   ws.current=s;
   s.onopen=()=>{
     setConnected(true);
     s.send(JSON.stringify({type:"set_station",stationId:station}));
     s.send(JSON.stringify({type:"set_mode",mode:"normal"}));
   };
   s.onclose=()=>setConnected(false);
   s.onerror=()=>setConnected(false);
   s.onmessage=e=>{
     const m=JSON.parse(e.data);
     if(m.type==="connected"){
       setStations(m.stations||[]);
       setCities(m.cityCatalog||[]);
       return;
     }
     if(m.type==="mode") setMode(m.mode);
     if(m.type==="selection"){
       setStation(m.selectedStation);
       setMode(m.mode||"normal");
     }
     if(m.type!=="decision") return;
     if(!m.observation){
       if(m.stationId===station) setLatest(x=>({...x,[station]:m}));
       return;
     }
     const id=m.observation.stationId;
     setLatest(x=>({...x,[id]:m}));
     setHistory(x=>({...x,[id]:[...(x[id]||[]),{
       time:new Date(m.observation.timestamp).toLocaleTimeString([], {hour:"2-digit",minute:"2-digit"}),
       temperature:m.observation.temperature,
       pressure:m.observation.pressure,
       humidity:m.observation.humidity
     }].slice(-40)}));
     if(id===station){
       setNearby(m.nearby||[]);
       setDepartments(m.departments||null);
       if(["FAULT SUSPECTED","REVIEW REQUIRED","WEATHER EVENT LIKELY","COMMUNICATION GAP"].includes(m.status) || m.observation?.sourceEvent!=="normal" || mode!=="normal"){
         setEvents(x=>({...x,[id]:[m,...(x[id]||[])].slice(0,20)}));
       }
     }
   };
 }

 function openCity(nameOrCity){
   const cityObj=typeof nameOrCity==="string" ? cities.find(c=>c.name===nameOrCity) : nameOrCity;
   if(!cityObj) return;
   const id=cityObj.stationId;
   setCity(cityObj.name);
   setStation(id);
   setTab("overview");
   setPage("dashboard");
   setLatest({});
   setHistory({});
   setEvents({});
   setNearby([]);
   setDepartments(null);
   if(ws.current?.readyState===WebSocket.OPEN){
     ws.current.send(JSON.stringify({type:"set_station",stationId:id}));
   }
 }

 function chooseStation(id){
   const s=stations.find(x=>x.id===id);
   if(!s) return;
   setStation(id);
   setCity(s.city);
   setMode("normal");
   setLatest(x=>({...x,[id]:undefined}));
   setHistory(x=>({...x,[id]:[]}));
   setEvents(x=>({...x,[id]:[]}));
   setNearby([]);
   setDepartments(null);
   ws.current?.send(JSON.stringify({type:"set_station",stationId:id}));
 }

 function chooseMode(v){setMode(v);ws.current?.send(JSON.stringify({type:"set_mode",mode:v}));}

 function startVoice(){
   const SR=window.SpeechRecognition||window.webkitSpeechRecognition;
   if(!SR){alert("Voice search is not supported by this browser. Try Chrome/Edge.");return;}
   if(voice){recognition.current?.stop();setVoice(false);return;}
   const r=new SR(); recognition.current=r; r.lang="en-IN"; r.interimResults=false;
   r.onstart=()=>setVoice(true); r.onend=()=>setVoice(false); r.onerror=()=>setVoice(false);
   r.onresult=e=>{
     const text=e.results[0][0].transcript.toLowerCase();
     const found=cities.find(c=>c.name.toLowerCase().includes(text)||text.includes(c.name.toLowerCase()));
     if(found){setSearch(found.name);openCity(found)} else setSearch(e.results[0][0].transcript);
   };
   r.start();
 }

 const current=latest[station];
 const currentHistory=history[station]||[];
 const currentEvents=events[station]||[];
 const cityStations=stations.filter(s=>s.city===city);
 if(page==="home") return <Home search={search} setSearch={setSearch} cities={cities} cityStrip={cityStrip} voice={voice} startVoice={startVoice} openCity={openCity}/>;
 return <Dashboard city={city} station={station} stationList={cityStations} connected={connected} mode={mode} tab={tab} setTab={setTab} current={current} history={currentHistory} nearby={nearby} departments={departments} events={currentEvents} chooseMode={chooseMode} chooseStation={chooseStation} back={()=>setPage("home")}/>;
}
